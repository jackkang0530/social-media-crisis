        function parseSelectedArticleData(cellData) {

            if (!cellData || cellData.trim() === '') return '';

            

            const trimmed = cellData.trim();

            

            if (trimmed.startsWith('(') && trimmed.endsWith(')')) {

                try {

                    const content = trimmed.slice(1, -1);

                    const parts = [];

                    let currentPart = '';

                    let inQuotes = false;

                    let quoteChar = '';

                    

                    for (let i = 0; i < content.length; i++) {

                        const char = content[i];

                        

                        if ((char === '"' || char === "'") && !inQuotes) {

                            inQuotes = true;

                            quoteChar = char;

                        } else if (char === quoteChar && inQuotes) {

                            inQuotes = false;

                            quoteChar = '';

                        } else if (char === ',' && !inQuotes) {

                            if (currentPart.trim()) {

                                parts.push(currentPart.trim());

                                currentPart = '';

                            }

                            continue;

                        }

                        

                        currentPart += char;

                    }

                    

                    if (currentPart.trim()) {

                        parts.push(currentPart.trim());

                    }

                    

                    if (parts.length > 0) {

                        let title = parts[0].trim();

                        if ((title.startsWith('"') && title.endsWith('"')) || 

                            (title.startsWith("'") && title.endsWith("'"))) {

                            title = title.slice(1, -1);

                        }

                        return title.trim();

                    }

                } catch (error) {

                    console.warn('Error parsing tuple data:', error, 'Original data:', cellData);

                }

            }

            

            return trimmed;

        }

        function parseCrisisCSVLine(line) {

            console.log('Parsing crisis line:', line);

            

            const csvParts = parseCSVLine(line);

            console.log('CSV parts:', csvParts);

            

            if (csvParts.length < 2) {

                console.warn('Crisis line has insufficient parts:', csvParts);

                return ['', '', '', ''];

            }

            

            const tupleString = csvParts[0];

            const link = csvParts[1].trim();

            

            let cleanTuple = tupleString;

            if (cleanTuple.startsWith('"') && cleanTuple.endsWith('"')) {

                cleanTuple = cleanTuple.slice(1, -1);

            }

            

            if (cleanTuple.startsWith('(') && cleanTuple.endsWith(')')) {

                cleanTuple = cleanTuple.slice(1, -1);

            }

            

            console.log('Clean tuple content:', cleanTuple);

            

            const parts = [];

            let currentPart = '';

            let inQuotes = false;

            let quoteChar = '';

            

            for (let i = cleanTuple.length - 1; i >= 0; i--) {

                const char = cleanTuple[i];

                

                if ((char === '"' || char === "'") && !inQuotes) {

                    inQuotes = true;

                    quoteChar = char;

                } else if (char === quoteChar && inQuotes) {

                    inQuotes = false;

                    quoteChar = '';

                } else if (char === ',' && !inQuotes) {

                    if (currentPart.trim()) {

                        parts.unshift(currentPart.trim());

                        currentPart = '';

                    }

                    continue;

                }

                

                currentPart = char + currentPart;

            }

            

            if (currentPart.trim()) {

                parts.unshift(currentPart.trim());

            }

            

            console.log('Parsed tuple parts:', parts);

            

            const cleanedParts = parts.map(part => {

                let cleaned = part.trim();

                if ((cleaned.startsWith('"') && cleaned.endsWith('"')) || 

                    (cleaned.startsWith("'") && cleaned.endsWith("'"))) {

                    cleaned = cleaned.slice(1, -1);

                }

                return cleaned.trim();

            });

            

            console.log('Cleaned parts:', cleanedParts);

            

            let articleTitle = '';

            let publication = '';

            let date = '';

            

            if (cleanedParts.length >= 3) {

                date = cleanedParts[cleanedParts.length - 1];

                publication = cleanedParts[cleanedParts.length - 2];

                articleTitle = cleanedParts.slice(0, cleanedParts.length - 2).join(', ');

            } else if (cleanedParts.length === 2) {

                publication = cleanedParts[0];

                date = cleanedParts[1];

                articleTitle = '';

            } else if (cleanedParts.length === 1) {

                articleTitle = cleanedParts[0];

            }

            

            const result = [articleTitle, publication, date, link];

            console.log('Final crisis parsing result:', result);

            

            return result;

        }

        function debounce(func, wait) {

            let timeout;

            return function executedFunction(...args) {

                const later = () => {

                    clearTimeout(timeout);

                    func(...args);

                };

                clearTimeout(timeout);

                timeout = setTimeout(later, wait);

            };

        }

        function applyFilters() {

            if (currentDataset !== '2' || currentClassification !== 'opb') {

                return;

            }

            

            filteredData = currentData.filter(row => {

                const giver = row[2]; // Giver column

                const receiver = row[3]; // Receiver column

                

                const giverMatch = currentGiverFilters.has('all') || currentGiverFilters.has(giver);

                const receiverMatch = currentReceiverFilters.has('all') || currentReceiverFilters.has(receiver);

                

                return giverMatch && receiverMatch;

            });

            

            currentPage = 1;

            updateTable();

            updateResultsInfo();

        }


        function updateTable() {

            console.log('Updating table with data:', filteredData.length, 'rows');

            

            const tableHeader = document.getElementById('tableHeader');

            const tableBody = document.getElementById('tableBody');

            

            // Update headers based on current dataset

            let headers = [];

            if (currentDataset === '1') {

                headers = ["Article Title", "Publication Title", "Date"];

            } else if (currentDataset === '2') {

                if (currentClassification === 'crisis') {

                    headers = ["Article Title", "Publication", "Date", "Link"];

                } else {

                    headers = ["Selected Article", "Stakeholder", "Giver", "Receiver", "Problematic Behavior", "Aftermath", "Platform", "Link"];

                }

            }

            

            // Create header row

            tableHeader.innerHTML = '<tr>' + 

                headers.map(header => `<th class="sortable">${header}</th>`).join('') + 

                '</tr>';

            

            // Calculate pagination

            const startIndex = (currentPage - 1) * itemsPerPage;

            const endIndex = Math.min(startIndex + itemsPerPage, filteredData.length);

            const pageData = filteredData.slice(startIndex, endIndex);

            

            // Create table rows

            if (pageData.length === 0) {

                tableBody.innerHTML = '<tr><td colspan="' + headers.length + '" class="loading">No data found</td></tr>';

            } else {

                tableBody.innerHTML = pageData.map((row, index) => {

                    const cells = row.map((cell, cellIndex) => {

                        let cellContent = cell || '';

                        

                        // Special handling for different cell types

                        if (headers[cellIndex] === 'Link' && cellContent.startsWith('http')) {

                            return `<td><a href="${cellContent}" target="_blank" class="link-cell">🔗 Link</a></td>`;

                        } else if (headers[cellIndex] === 'Problematic Behavior') {

                            const colorClass = getBehaviorColorClass(cellContent);

                            return `<td><span class="behavior-tag ${colorClass}">${cellContent}</span></td>`;

                        } else {

                            return `<td title="${cellContent}" onclick="showCellContent('${headers[cellIndex]}', '${cellContent.replace(/'/g, "&#39;")}')">${cellContent}</td>`;

                        }

                    }).join('');

                    

                    return `<tr>${cells}</tr>`;

                }).join('');

            }

            

            updatePagination();

        }
